cmake_minimum_required(VERSION 3.16)

project(MyProject)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# ftxui was removed: nothing in this project (src/ or include/) references it -
# only raylib is actually used for the front end. Requiring it would just
# make the build fail on any machine that doesn't happen to have it installed,
# for no benefit. Re-add it here if you start using it for something.
find_package(raylib REQUIRED)

add_executable(MyProject
    src/main.cpp

    # board
    src/board/board.cpp
    src/board/space.cpp

    # cards
    src/card/Card.cpp
    src/card/Effect.cpp
    src/card/CardText.cpp

    # game
    src/game/game.cpp
    src/game/Player.cpp

    # fighters
    src/fighter/Heroes.cpp
    src/fighter/Sidekick.cpp
    src/fighter/Dracula.cpp
    src/fighter/Sisters.cpp
    src/fighter/SherlockHolmes.cpp
    src/fighter/Dr_Watson.cpp
    src/fighter/Invisibleman.cpp
    src/fighter/InvisibleManToken.cpp

    # front end
    src/front/front.cpp
    src/front/main_menu.cpp
    src/front/age_page.cpp
    src/front/choose.cpp
    src/front/game_page.cpp
    src/front/combat_page.cpp
    src/front/handmenu.cpp
    src/front/CardViewWindow.cpp
    src/front/DeckCardWindow.cpp
)

target_include_directories(MyProject PRIVATE include include/front)

target_link_libraries(MyProject PRIVATE
    raylib
)
